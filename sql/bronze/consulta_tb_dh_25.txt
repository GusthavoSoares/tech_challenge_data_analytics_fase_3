--Análise da base tb_dh25--
-- 1. Quantidade de registros
SELECT COUNT(*) AS qtd_registros_25
FROM tb_dh_25;
-- 3495 Respostas da pesquisa--

-- 2. Quantidade de colunas (não perguntas)
SELECT COUNT(*) AS qtd_colunas_25
FROM information_schema.columns
WHERE table_name = 'tb_dh_25';
-- 388 colunas (nao perguntas)--

-- 3. Registros sem ID
SELECT
    COUNT(*) AS total_registros,
    SUM(
        CASE
            WHEN "0.a_token" IS NULL
              OR TRIM(CAST("0.a_token" AS VARCHAR)) = ''
            THEN 1
            ELSE 0
        END
    ) AS ids_nulos
FROM tb_dh_25;
-- Não há nenhuma linha com ID nulo/vazio--

-- 4. Verificação de ID duplicado
SELECT
    "0.a_token" AS id,
    COUNT(*) AS quantidade
FROM tb_dh_25
WHERE "0.a_token" IS NOT NULL
GROUP BY "0.a_token"
HAVING COUNT(*) > 1
ORDER BY quantidade DESC;
-- um ID duplicado: tdm01ed14n10hdl0rkrtdm01edwge73j

-- 5. Verificação se a quantidade de IDs únicos
-- é a mesma quantidade de registros
SELECT
    COUNT(*) AS total_registros,
    COUNT(DISTINCT "0.a_token") AS ids_unicos
FROM tb_dh_25;
-- não é igual devido ao ID duplicado

-- 6. Verificação dos tipos de variáveis
SELECT
    data_type,
    COUNT(*) AS quantidade_colunas
FROM information_schema.columns
WHERE table_schema = 'default'
  AND table_name = 'tb_dh_25'
GROUP BY data_type
ORDER BY quantidade_colunas DESC;
-- Todas variáveis varchar--

--Atráves de Análise em Excel, são 69 perguntas distintas e a coluna de ID--

-- 1. Distribuição por faixa etária
SELECT
    "1.a.1_faixa_idade" AS faixa_idade,
    COUNT(*) AS quantidade,
    ROUND(
        100.0 * COUNT(*) / SUM(COUNT(*)) OVER (),
        2
    ) AS percentual
FROM tb_dh_25
WHERE "1.a.1_faixa_idade" IS NOT NULL
GROUP BY "1.a.1_faixa_idade"
ORDER BY quantidade DESC;


-- Mediana da idade
SELECT
    approx_percentile(
        TRY_CAST("1.a_idade" AS DOUBLE),
        0.5
    ) AS idade_mediana
FROM tb_dh_25
WHERE TRY_CAST("1.a_idade" AS INTEGER) IS NOT NULL;
-- A idade mediana dos partipantes da pesquisa é de 32 anos, e a faixa de idade com maiores participantes é de 30-34 anos (27,5%)--

-- 15. Uso básico de IA
-- Em 2025 a pergunta passou de 4.m para 4.j
SELECT
    CASE
        WHEN "4.j_usa_chatgpt_ou_copilot_no_trabalho?" =
             'Não utilizo nenhum tipo de solução de IA Generativa para melhorar a produtividade no dia a dia.'
        THEN 'Não utiliza IA Generativa'
        ELSE 'Utiliza IA Generativa'
    END AS uso_ia,
    COUNT(*) AS quantidade,
    ROUND(
        100.0 * COUNT(*) / SUM(COUNT(*)) OVER (),
        2
    ) AS percentual
FROM tb_dh_25
WHERE "4.j_usa_chatgpt_ou_copilot_no_trabalho?" IS NOT NULL
GROUP BY
    CASE
        WHEN "4.j_usa_chatgpt_ou_copilot_no_trabalho?" =
             'Não utilizo nenhum tipo de solução de IA Generativa para melhorar a produtividade no dia a dia.'
        THEN 'Não utiliza IA Generativa'
        ELSE 'Utiliza IA Generativa'
    END
ORDER BY quantidade DESC;
--baseado nessa classificação de uso ou nao de IA, temos que 98,88% dos profissionais usam IAs em alguma parte do trabalho
